#!/bin/awk -f
BEGIN{ FS="[: ]";}
{
    if (NF == 9) {
        LAST = $8
        TOT = ($6+$7+$8);
        AVG = (TOT/3);
    } else {
        LAST = $7
        TOT = ($5+$6+$7);
        AVG = TOT/3;
    }
    if ($2 == "Took" || $3 == "Took"){
        print $1, "$"TOT;
    }
    if(NF == 9 && (LAST >= 10 && LAST <= 200)) {
        print $1,$2,$3, "$"LAST;
    } else if (NF == 8 && (LAST >= 10 && LAST <= 200)) {
        print $1,$2,"$"LAST;
    }
    if (AVG < 300) {
        if (NF == 9) {
            print $1,$2,$3, "$"AVG;
        } else {
            print $1,$2, "$"AVG;
        }
    }
}
